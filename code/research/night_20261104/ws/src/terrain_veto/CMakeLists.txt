cmake_minimum_required(VERSION 3.8)
project(terrain_veto)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-O2 -Wall -Wextra)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
add_executable(terrain_veto_node src/terrain_veto_node.cpp)
ament_target_dependencies(terrain_veto_node rclcpp sensor_msgs nav_msgs)
install(TARGETS terrain_veto_node DESTINATION lib/${PROJECT_NAME})
ament_package()
